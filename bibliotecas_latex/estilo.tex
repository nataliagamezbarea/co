}

% --- A: MARCADOR (mayúsculas gruesas sobre una franja resaltada, estilo img.1)
\newcommand{\tituloA}[2]{%
  \begin{center}
  \begin{tikzpicture}
    \node (t) {{\markerfont\fontsize{34}{34}\selectfont\color{ink} #1}};
    \begin{pgfonlayer}{background}
      \fill[tertiary, rotate around={-1.4:($(t.south west)!0.5!(t.north east)$)}]
        ($(t.south west)+(-0.55,-0.20)$) rectangle ($(t.north east)+(0.55,0.24)$);
    \end{pgfonlayer}
  \end{tikzpicture}
  \end{center}
  \vspace{-1mm}
  \begin{center}\textcolor{accent}{\rule{2.6cm}{2.6pt}}\end{center}
  \vspace{2.5mm}
  \begin{center}{\headingfontxb\fontsize{14}{14}\selectfont\color{primary} #2}\end{center}
  \vspace{6mm}
}
